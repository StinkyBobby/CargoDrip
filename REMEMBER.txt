        ↓
[ api (роуты) ]
        ↓
[ service (логика) ]
        ↓
[ repositories (база данных) ]
        ↓
[ models (таблицы SQLAlchemy) ]

schemas используются на входе и выходе в api и service.
service — мозг
repositories — делают запросы к базе.
models — таблицы



Создание пользователей для PGADMIN

Для вставки пользователей

INSERT INTO employee (username, password_hash, role, email)
VALUES (
  'admin',
  '$2b$12$uK3ZQ9eYvQvYJZJ7ZxkU6u1zqZzjz0xZzjzjz0xZzjzjz0xZzjzjz',  -- сюда bcrypt-хэш
  'admin',  -- или 'driver'

);

Изменение пароля
UPDATE employee
SET password_hash = '$2b$12$новый_хэш'
WHERE username = 'admin';

Удаление пользователя
DELETE FROM employee WHERE username = 'admin';
DROP ROLE admin_user;
